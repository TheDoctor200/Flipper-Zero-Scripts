DELAY 5000

DELAY 800
STRING cmd
ENTER
DELAY 1200
STRING cd "%USERPROFILE%\Desktop"
ENTER

ENTER

ENTER

ENTER
STRING echo This organization was created by Flipper Zero! >> README.txt
ENTER
DELAY 500
STRING notepad README.txt
ENTER
